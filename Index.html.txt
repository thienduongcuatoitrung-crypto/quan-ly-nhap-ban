  <div id="tongThu">Gốc: 0, Lãi: 0</div>
</div>

<script>
let ds = [];

function themHopDong(){
  const ten=document.getElementById("ten").value;
  const tien=parseFloat(document.getElementById("tien").value);
  const lai=parseFloat(document.getElementById("lai").value);
  const ngay=document.getElementById("ngay").value;
  const chuKy=parseInt(document.getElementById("chuKy").value);
  if(!ten||!tien||!lai||!ngay||!chuKy){alert("Nhập đủ thông tin"); return;}
  ds.push({ten,tien,lai,ngay,chuKy,daThu:false});
  hienThi();
}

function hienThi(){
  const div=document.getElementById("danhSachHopDong");
  div.innerHTML="";
  ds.forEach((h,i)=>{
    const today=new Date().toISOString().slice(0,10);
    const overdue=!h.daThu && h.ngay < today;
    div.innerHTML+=`<div style="background:${overdue?'#ffcccc':'#e8e8e8'}">
      <b>${h.ten}</b> | 💰${h.tien} | 📈${h.lai}% | 📅${h.ngay}
      <button onclick="daThu(${i})">Đã thu</button>
    </div>`;
  });
  tinhTongThu();
}

function daThu(i){ds[i].daThu=true;hienThi();}

function tinhTongThu(){
  let tongGoc=0,tongLai=0;
  ds.forEach(h=>{
    if(h.daThu){
      tongGoc+=h.tien;
      tongLai+=h.tien*h.lai/100;
    }
  });
  document.getElementById("tongThu").innerText=`Gốc: ${tongGoc}, Lãi: ${tongLai}`;
}

setInterval(()=>{
  const today=new Date().toISOString().slice(0,10);
  ds.forEach(h=>{
    if(!h.daThu && h.ngay===today){alert("🔔 Đến hạn thu tiền: "+h.ten);}
  });
},60000);

hienThi();
</script>
